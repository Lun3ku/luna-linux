# luna-base is the base system and the reliability layer of Luna Linux.
# luna-release carries the system branding. It is a separate package so that
# it can go onto the boot image without 307 MB of base-devel.
luna-release
# luna-keyring holds the key the luna-* packages are signed with. Without it
# the system cannot verify the signature of its own updates after a reboot.
luna-keyring
#
# The kernel is deliberately NOT listed here. The user picks linux or
# linux-zen in the installer, and the headers must match the chosen kernel
# (linux-headers for linux, linux-zen-headers for linux-zen). Hard-coding
# linux-headers here would land headers from the wrong kernel on a linux-zen
# system. The installer installs the kernel and its headers as a pair.

# --- the base ---
base                 # base system metapackage (the counterpart of linux-base in Debian)
base-devel           # compilers and makepkg: nothing from the AUR builds without them
linux-firmware       # hardware firmware
mkinitcpio           # initramfs generator, not part of base
sudo
git
nano                 # a simple editor, has to be there always
vim                  # vi-compatible, for muscle memory
networkmanager

# --- reliability: snapshots and the bootloader ---
btrfs-progs          # the filesystem with snapshots
snapper              # snapshot management
snap-pac             # takes a snapshot automatically before every pacman operation
grub                 # bootloader (systemd-boot cannot boot into a snapshot)
grub-btrfs           # a GRUB menu entry for every snapshot
inotify-tools        # grub-btrfsd needs it to watch the snapshots
efibootmgr

# --- reliability: memory and maintenance ---
plymouth             # a graphical splash instead of a stream of kernel text
                     # while booting; the luna theme ships in this package
zram-generator       # compressed swap in RAM instead of a swap file
pacman-contrib       # paccache, the automatic package cache cleanup
reflector            # automatic mirror list refresh
