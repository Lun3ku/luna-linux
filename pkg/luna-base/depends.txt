# luna-base — база системы и слой надёжности Luna Linux.
# luna-release — брендинг системы. Вынесен отдельным пакетом, чтобы его
# можно было положить на загрузочный образ без 307 МБ base-devel.
luna-release
# luna-keyring — ключ, которым подписаны пакеты luna-*. Без него система
# после перезагрузки не сможет проверить подпись собственных обновлений.
luna-keyring
#
# Ядро здесь сознательно НЕ указано. Пользователь выбирает linux или
# linux-zen в установщике, а headers обязаны соответствовать выбранному
# ядру (linux-headers к linux, linux-zen-headers к linux-zen). Зашей мы
# linux-headers сюда — на linux-zen приехали бы заголовки не от того ядра.
# Пару «ядро + headers» ставит установщик.

# --- база ---
base                 # метапакет базовой системы (аналог linux-base в Debian)
base-devel           # компиляторы и makepkg: без них не собрать ничего из AUR
linux-firmware       # прошивки железа
mkinitcpio           # генератор initramfs, в base не входит
sudo
git
nano                 # простой редактор, должен быть всегда
vim                  # vi-совместимый, мышечная память
networkmanager

# --- надёжность: снапшоты и загрузчик ---
btrfs-progs          # файловая система со снапшотами
snapper              # управление снапшотами
snap-pac             # снапшот автоматически перед каждой операцией pacman
grub                 # загрузчик (systemd-boot не умеет грузиться в снапшот)
grub-btrfs           # пункты меню GRUB для каждого снапшота
inotify-tools        # нужен grub-btrfsd для слежения за снапшотами
efibootmgr

# --- надёжность: память и обслуживание ---
plymouth             # графическая заставка вместо потока текста ядра
                     # при загрузке; тема luna идёт в этом же пакете
zram-generator       # сжатый swap в ОЗУ вместо файла подкачки
pacman-contrib       # paccache — автоочистка кэша пакетов
reflector            # автообновление списка зеркал
