# luna-desktop is Hyprland and everything without which it is a bare window
# manager rather than a desktop.
luna-release                  # branding; base-devel is not dragged in here

# --- compositor and session ---
hyprland
uwsm                          # starts Hyprland as a real systemd session:
                              # it carries things through to
                              # graphical-session.target, after which the units
                              # below start and restart on their own
xdg-desktop-portal-hyprland   # screen sharing, file dialogs
hyprpolkitagent               # graphical privilege prompts
hyprcursor

# --- desktop shell ---
waybar                        # the panel
rofi-wayland                  # launching programs, switching windows, menus
mako                          # notifications
swaybg                        # the wallpaper. It draws through cairo and
                              # wl_shm, so it works without a GPU too.
                              # hyprpaper 0.8.4 crashes with SIGSEGV under the
                              # same conditions - verified
hyprpaper                     # kept as an alternative for those who have a GPU
                              # and want IPC control over the wallpaper
hyprlock                      # screen lock
hypridle                      # automatic lock and screen blanking when idle
hyprshutdown                  # the shutdown menu
hyprsunset                    # night mode, warmer colours in the evening
hyprpicker                    # colour picker

# --- clipboard and screenshots ---
wl-clipboard
cliphist                      # clipboard history
grim                          # screen capture
slurp                         # region selection
swappy                        # annotating a capture

# --- logging in ---
greetd
greetd-tuigreet               # a login screen in the same style as the installer

# --- applications ---
kitty                         # terminal
thunar                        # file manager
thunar-volman                 # automatic mounting of removable media
tumbler                       # file thumbnails
gvfs                          # access to disks and network from the file manager
xdg-user-dirs
polkit

# --- sound ---
pipewire
pipewire-pulse                # provides the PulseAudio interface; pulseaudio
                              # itself conflicts with it and breaks screen
                              # capture
pipewire-alsa
wireplumber
pavucontrol

# --- network and bluetooth ---
network-manager-applet
bluez
bluez-utils
blueman

# --- hardware odds and ends ---
brightnessctl                 # screen brightness
playerctl                     # media control from the keyboard

# --- appearance ---
# The font is split deliberately. The ttf-jetbrains-mono-nerd package weighs
# 232 MiB: 96 files, every weight of every variant. An ordinary font plus a
# symbol font give exactly the same result for 10 MiB, because fontconfig picks
# the icons out of Symbols Nerd Font Mono by itself. Verified on every glyph
# in use.
ttf-jetbrains-mono            # the main monospace face
ttf-nerd-fonts-symbols-mono   # panel icons, substituted automatically
noto-fonts
noto-fonts-emoji
noto-fonts-cjk
papirus-icon-theme
qt5-wayland
qt6-wayland
qt6ct
